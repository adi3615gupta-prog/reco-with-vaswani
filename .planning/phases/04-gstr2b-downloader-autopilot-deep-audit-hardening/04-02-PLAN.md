         export function removeAllMergesFromWorkbook(wb) {
             if (!wb || !wb.SheetNames) return wb;
             for (const name of wb.SheetNames) {
                 const ws = wb.Sheets[name];
                 if (ws) {
                     ws['!merges'] = [];
                     delete ws['!merges'];
                 }
             }
             return wb;
         }

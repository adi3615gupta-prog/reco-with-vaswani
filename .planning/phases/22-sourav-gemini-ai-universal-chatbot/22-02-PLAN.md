### T1 — Create `src/lib/geminiApi.ts` — Frontend API wrapper
- **File:** `src/lib/geminiApi.ts` (new)
- **Content:**
  ```typescript
  // src/lib/geminiApi.ts
  import { getApiBase } from './api';
  
  export interface ChatMessage {
    role: 'user' | 'model';
    text: string;
  }
  
  export interface GeminiContext {
    clientName?: string;
    gstin?: string;
    fy?: string;
    module?: string;
  }
  
  export async function sendChatMessage(
    message: string,
    context: GeminiContext = {},
    history: ChatMessage[] = []
  ): Promise<string> {
    const res = await fetch(`${getApiBase()}/api/gemini/chat`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message, context, history }),
    });
    const data = await res.json();
    if (!res.ok) throw new Error(data.error || 'Gemini API error');
    return data.reply as string;
  }
  
  export async function getGeminiStatus(): Promise<{ configured: boolean }> {
    const res = await fetch(`${getApiBase()}/api/gemini/status`);
    return res.json();
  }
  
  export async function saveGeminiApiKey(apiKey: string): Promise<void> {
    const res = await fetch(`${getApiBase()}/api/gemini/config`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ apiKey }),
    });
    if (!res.ok) throw new Error('Failed to save API key');
  }
  ```

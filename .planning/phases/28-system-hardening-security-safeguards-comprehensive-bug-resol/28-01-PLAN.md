<threat_model>
<threat>
  <id>T-28-01</id>
  <description>Unauthenticated exfiltration of taxpayer GST portal credentials over local network.</description>
  <mitigation>Require JWT authentication on `/api/ims/credentials` and project `portal_password` as a masked boolean flag.</mitigation>
</threat>
<threat>
  <id>T-28-02</id>
  <description>Cross-origin drive-by attacks via reflected CORS origin header.</description>
  <mitigation>Validate origins against localhost and RFC 1918 private subnets; reject arbitrary external domains.</mitigation>
</threat>
<threat>
  <id>T-28-03</id>
  <description>Arbitrary command injection through folder launcher shell concatenation.</description>
  <mitigation>Replace `child_process.exec` with `child_process.spawn('explorer.exe', [path], { shell: false })`.</mitigation>
</threat>
<threat>
  <id>T-28-04</id>
  <description>Arbitrary local file disclosure through unvalidated spreadsheet path parameter.</description>
  <mitigation>Validate that target paths resolve within the configured application storage directory.</mitigation>
</threat>
</threat_model>

    5. Reject all foreign internet origins.
  </description>
